#THE FIRST FOURTEEN
news_event = {
	id = mongolia_news_events.2
	title = mongolia_news_events.2.t
	desc = mongolia_news_events.2.desc
	picture = GFX_news_event_bombogor_nogoon

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			every_army_leader = {
				if = {
					limit = {
						not = { has_character_flag = MON_protected }
						has_trait = trait_MON_nationalist_sympathies
					}
					random = {
						chance = 90
						set_character_flag = MON_14_trial_target
					}
				}
				else_if = {
					limit = {
						not = { has_character_flag = MON_protected }
						has_trait = trait_MON_stalinist_loyalties 
					}
					random = {
						chance = 25
						set_character_flag = MON_14_trial_target
					}
				}
				else_if = {
					limit = {
						not = { has_character_flag = MON_protected }
					}
					random = {
						chance = 75
						set_character_flag = MON_14_trial_target
					}
				}
			}
		}
	}

	option = {
		name = mongolia_news_events.2.a
		every_army_leader = {
			limit = { has_character_flag = MON_14_trial_target }
			custom_effect_tooltip = MON_14_trial_tt
			hidden_effect = {
				retire = yes
			}
		}
	}
}
