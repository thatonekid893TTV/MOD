	desc = romania.1.d

		name = romania.1.a

			factor = 90
			modifier = { # Don't accept if in a faction that can protect them
				factor = 0

					any_other_country = {
						is_major = yes
						NOT = { tag = SOV }
						OR = {
							is_in_faction_with = ROM
							has_guaranteed = ROM
						}
					}

			}

		  limit = { has_any_power_balance = yes  }
		  add_power_balance_value = {
		      id = ROM_power_balance
		      value = 0.3
	        }
		}
		FROM = {
			country_event = { days = 1 id = soviet.4 }
		}
		set_country_flag = bessarabia_ceded
	}

	option = { # Refuse
		name = romania.1.b
		ai_chance = {
			base = 0
			modifier = {
				add = 10
				is_in_faction = yes
				NOT = { is_in_faction_with = FROM }
			}

			modifier = {
				add = 100
				has_war_with = FROM
			}
			modifier = {
				is_historical_focus_on = yes
				NOT = {has_war_with = FROM}
				factor = 0
			}
			modifier = {
				factor = 0
				is_in_faction_with = FROM
			}
		}
		if = {
		  limit = { has_any_power_balance = yes  }
		 add_power_balance_value = {
		      id = ROM_power_balance
		      value = -0.3
	        }
