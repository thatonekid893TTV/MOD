add_namespace = useful-events-quarior-coup

country_event = {
	id = useful-events-quarior-coup.1
	title = useful-events-quarior-coup.1.t
	desc = useful-events-quarior-coup.1.desc
	picture = GFX_report_event_gre_city_devastated
	timeout_days = 10
	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			FROM = {
				save_event_target_as = country_do_coup
			}
		}
	}
	option = {
		name = useful-events-quarior-coup.1.a
		
		random_list = {
			1 = {
				# Success Coup
				
				modifier = {
					factor = 2
					AND = {
						has_war_with = FROM
						has_capitulated = yes
					}
				}
				modifier = {
					factor = 0.5
					set_temp_variable = { ideology_v = FROM.GetRulingIdeology }
					set_temp_variable = { ideology_pop_v = party_popularity@var:ideology_v }
					check_variable = {
						ideology_pop_v < 0.05
					}
				}
				modifier = {
					factor = 10
					is_subject_of = FROM
				}
				modifier = {
					factor = 0.5
					has_stability > 0.6
				}
				
				news_event = { id = useful-events-quarior-coup.2 days = 1 random_days = 1 }
				effect_tooltip = {
					set_temp_variable = { from_v = FROM }
					set_politics = { ruling_party = var:from_v }
					add_popularity = {
						ideology = var:from_v
						popularity = 0.1
					}
					var:from_v = { 
						puppet = {
							target = ROOT
							end_civil_wars = no
						}
					}
				}
			}
			1 = {	
				# Fail Coup
				modifier = {
					factor = 0.5
					set_temp_variable = { ideology_v = FROM.GetRulingIdeology }
					set_temp_variable = { ideology_pop_v = party_popularity@var:ideology_v }
					check_variable = {
						ideology_pop_v > 0.25
					}
				}
				
				set_temp_variable = {
					chance_civil_war_v = 1
				}
				
				subtract_from_temp_variable = {
					var = chance_civil_war_v
					value = ROOT.stability
				}
				
				multiply_temp_variable = {
					var = chance_civil_war_v
					value = 100
				}
				
				random = {
					chance = var:chance_civil_war_v
					
					start_civil_war = {
						ideology = FROM
						set_temp_variable = { this_v = THIS }
						FROM = { 
							puppet = {
								target = var:this_v
								end_civil_wars = no
							}
						}
					}
				}
				
				FROM = { country_event = { id = useful-events-quarior-coup.3 days = 1 random_days = 1 } }
			}
		}
	}
}
