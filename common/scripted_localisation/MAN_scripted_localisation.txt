# Allows you to create your own dynamic keys
# to be called in localization.
# defined_text -> this is it, we're defining the text
# text -> a discrete entry that can be picked to display in loc.
# trigger -> determines if a text entry will be picked or not.
# (The triggers need to be valid for the scope the key is called in
# (eg Root or From.From).)
# localization_key -> points to the localization key
# that'll be used if trigger passes

defined_text = { # Country
	name = MAN_socialist_coup_risk
	text = {
		trigger = {
			check_variable = { man_soviet_influence < 0.08 }
			NOT = { has_completed_focus = MAN_socialist_victory }
		}
		localization_key = MAN_risk_low
	}
	text = {
		trigger = {
			check_variable = { man_soviet_influence < 0.17 }
			NOT = { has_completed_focus = MAN_socialist_victory }
		}
		localization_key = MAN_risk_medium
	}
	text = {
		trigger = {
			check_variable = { man_soviet_influence < 0.19 }
			NOT = { has_completed_focus = MAN_socialist_victory }
		}
		localization_key = MAN_risk_high
	}
	text = {
		trigger = {
			check_variable = { man_soviet_influence < 0.25 }
			NOT = { has_completed_focus = MAN_socialist_victory }
		}
		localization_key = MAN_risk_imminent
	}
	text = {
		trigger = {
			has_completed_focus = MAN_socialist_victory
		}
		localization_key = MAN_risk_succeed
	}
	text = {
		localization_key = MAN_risk_null
	}
}
