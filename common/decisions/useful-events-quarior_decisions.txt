	debug_quarior_create_dynamic_country = {
		fire_only_once = no

		}
	}
	
	debug_quarior_annex_dynamic_country = {
		fire_only_once = no
		
		complete_effect = {
			set_temp_variable = {
				from_v = ROOT
			}
			every_other_country = {
				limit = {
					original_tag = ROOT
				}
				set_temp_variable = {
					this_country = THIS
				}
				var:this_country = {
					every_character = {
						set_nationality = var:from_v
					}
					transfer_navy = {
						target = ROOT
					}
					send_equipment_fraction = {
						value = 1
						target = var:from_v
					}
				}
				every_country = {
					limit = {
						is_subject_of = var:this_country
					}
					var:from_v = {
						puppet = PREV
					}
				}
				var:from_v = {
					annex_country = {
						target = PREV
						transfer_troops = yes
					}
				}
				var:this_country = {
					every_core_state = {
						add_core_of = var:from_v
					}
					every_state = {
						limit = {
							is_claimed_by = var:this_country
						}
						add_claim_by = var:from_v
					}
				}
			}
		}
		
		visible = {
			any_other_country = {
				original_tag = ROOT
				exists = yes
			}
		}
	}
