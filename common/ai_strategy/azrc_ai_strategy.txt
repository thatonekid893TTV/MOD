AZR_vol_2aif = {
    allowed = {
        always = yes
    }
    enable = {
		AST = { has_idea = MMR_AST_2aif_idea }
		NOT = {
			tag = AST
			has_war_with = AST
		}
        OR = {
			has_government = democratic
			has_government = neutrality
		}
		any_owned_state = {
			OR = {
				is_on_continent = asia
				is_on_continent = australia
			}
		}
		NOT = { is_in_faction_with = AST }
    }
    abort = {
        AST = { NOT = { has_idea = MMR_AST_2aif_idea } }
    }
    ai_strategy = {
		type = diplo_action_acceptance
		id = NZL
		value = 100
		target = send_volunteers
	}
}
